⚙️ Setup Guide

Follow these steps once to connect Cloudinary & Google Sheets

1

Create a Cloudinary Account

Go to cloudinary.com → Sign up free → From your Dashboard, copy your Cloud Name. Then go to Settings → Upload → Upload Presets → Add a new preset, set it to Unsigned, name it lakyaa_upload.

2

Create your Google Sheet

Create a new Google Sheet. Add two tabs (sheets): one named Gallery and one named Testimonials.

For Gallery, set row 1 headers:

Date | Title | Category | ImageURL | PublicID

For Testimonials, set row 1 headers:

Date | Name | Service | Rating | Message | PhotoURL
3

Deploy the Google Apps Script (Web App)

In your Google Sheet: Extensions → Apps Script. Paste the code below, then Deploy → New Deployment → Web App → set "Who has access" to Anyone → Copy the Web App URL.

function doPost(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const data = JSON.parse(e.postData.contents); const sheet = ss.getSheetByName(data.sheet); if (!sheet) return ContentService.createTextOutput(JSON.stringify({status:'error',msg:'Sheet not found'})).setMimeType(ContentService.MimeType.JSON); sheet.appendRow(data.row); return ContentService .createTextOutput(JSON.stringify({status:'ok'})) .setMimeType(ContentService.MimeType.JSON); } function doGet(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheetName = e.parameter.sheet; const callback = e.parameter.callback; const sheet = ss.getSheetByName(sheetName); if (!sheet) { const out = JSON.stringify([]); return ContentService.createTextOutput(callback ? callback + '(' + out + ')' : out).setMimeType(ContentService.MimeType.JAVASCRIPT); } const rows = sheet.getDataRange().getValues(); const headers = rows[0]; const records = rows.slice(1).map(r => { const obj = {}; headers.forEach((h,i) => obj[h] = r[i]); return obj; }); const json = JSON.stringify(records); // Support JSONP for cross-origin requests const output = callback ? callback + '(' + json + ')' : json; const mime = callback ?
